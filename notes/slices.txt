1. Compact (Consecutive Remove Duplicates):
-------------------------------------------
package main

import (
	"fmt"
	"slices"
)

func main() {
	seq := []int{0, 1, 1, 2, 3, 5, 8}
	seq = slices.Compact(seq)
	fmt.Println(seq)
}

O/P: [0 1 2 3 5 8]

2. Compare (To Check two slices are equal):
------------------------------------------
package main

import (
	"fmt"
	"slices"
)

func main() {
	s1 := []int{1, 2, 3}
	s2 := []int{1, 2, 3}
	fmt.Println(slices.Compare(s1, s2))
}

O/P: 0

3. Equal:
---------
package main

import (
	"fmt"
	"slices"
)

func main() {
	s1 := []int{1, 2, 3}
	s2 := []int{1, 2, 3}
	fmt.Println(slices.Equal(s1, s2))
}


4. Concat:
----------
package main

import (
	"fmt"
	"slices"
)

func main() {
	s1 := []int{0, 1, 2, 3}
	s2 := []int{4, 5, 6}
	concat := slices.Concat(s1, s2)
	fmt.Println(concat)
}
